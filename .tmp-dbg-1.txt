#!/usr/bin/env bash
set -euo pipefail
cd "$(git rev-parse --show-toplevel)"
LOG="$(git rev-parse --show-toplevel)/batch-14.log"
: > "$LOG"

commit() {
  local msg="$1"; shift
  local want="" got="" line=""
  git add -- "$@"
  # Assert staged-set == requested set (no strays, all present)
  want=$(printf '%s\n' "$@" | sort)
  got=$(git diff --cached --name-only | sort)
  if [ "$got" != "$want" ]; then
    echo "MISMATCH on: $msg" | tee -a "$LOG"
    echo "  want:" | tee -a "$LOG"; echo "$want" | sed 's/^/    /' | tee -a "$LOG"
    echo "  got :" | tee -a "$LOG"; echo "$got" | sed 's/^/    /' | tee -a "$LOG"
    exit 1
  fi
  git commit -q -m "$msg"
